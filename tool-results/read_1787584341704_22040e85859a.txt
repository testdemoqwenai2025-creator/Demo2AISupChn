   236→function ParticleCanvas() {
   237→  const canvasRef = useRef<HTMLCanvasElement>(null)
   238→  
   239→  useEffect(() => {
   240→    const canvas = canvasRef.current
   241→    if (!canvas) return
   242→    
   243→    const ctx = canvas.getContext('2d')
   244→    if (!ctx) return
   245→    
   246→    let animationId: number
   247→    const particles: Array<{
   248→      x: number
   249→      y: number
   250→      size: number
   251→      speedX: number
   252→      speedY: number
   253→      opacity: number
   254→      pulse: number
   255→    }> = []
   256→    
   257→    const resize = () => {
   258→      canvas.width = window.innerWidth
   259→      canvas.height = window.innerHeight
   260→    }
   261→    
   262→    resize()
   263→    window.addEventListener('resize', resize)
   264→    
   265→    const particleCount = Math.min(Math.floor(window.innerWidth * 0.06), 80)
   266→    
   267→    for (let i = 0; i < particleCount; i++) {
   268→      particles.push({
   269→        x: Math.random() * canvas.width,
   270→        y: Math.random() * canvas.height,
   271→        size: Math.random() * 1.5 + 0.5,
   272→        speedX: (Math.random() - 0.5) * 0.3,
   273→        speedY: (Math.random() - 0.5) * 0.3,
   274→        opacity: Math.random() * 0.5 + 0.1,
   275→        pulse: Math.random() * Math.PI * 2
   276→      })
   277→    }
   278→    
   279→    const drawConnections = () => {
   280→      for (let i = 0; i < particles.length; i++) {
   281→        for (let j = i + 1; j < particles.length; j++) {
   282→          const dx = particles[i].x - particles[j].x
   283→          const dy = particles[i].y - particles[j].y
   284→          const dist = Math.sqrt(dx * dx + dy * dy)
   285→          
   286→          if (dist < 120) {
   287→            const opacity = (1 - dist / 120) * 0.08
   288→            ctx.beginPath()
   289→            ctx.moveTo(particles[i].x, particles[i].y)
   290→            ctx.lineTo(particles[j].x, particles[j].y)
   291→            ctx.strokeStyle = `rgba(16, 185, 129, ${opacity})`
   292→            ctx.lineWidth = 0.5
   293→            ctx.stroke()
   294→          }
   295→        }
   296→      }
   297→    }
   298→    
   299→    const animate = () => {
   300→      ctx.clearRect(0, 0, canvas.width, canvas.height)
   301→      
   302→      particles.forEach(p => {
   303→        p.x += p.speedX
   304→        p.y += p.speedY
   305→        p.pulse += 0.02
   306→        
   307→        if (p.x < 0 || p.x > canvas.width || p.y < 0 || p.y > canvas.height) {
   308→          p.x = Math.random() * canvas.width
   309→          p.y = Math.random() * canvas.height
   310→        }
   311→        
   312→        const o = p.opacity + Math.sin(p.pulse) * 0.15
   313→        ctx.beginPath()
   314→        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2)
   315→        ctx.fillStyle = `rgba(16, 185, 129, ${o})`
   316→        ctx.fill()
   317→      })
   318→      
   319→      drawConnections()
   320→      animationId = requestAnimationFrame(animate)
   321→    }
   322→    
   323→    animate()
   324→    
   325→    return () => {
   326→      window.removeEventListener('resize', resize)
   327→      cancelAnimationFrame(animationId)
   328→    }
   329→  }, [])
   330→  
   331→  return (
   332→    <canvas
   333→      ref={canvasRef}
   334→      className="fixed inset-0 pointer-events-none z-0"
   335→      style={{ opacity: 0.6 }}
   336→    />
   337→  )
   338→}

   416→        {/* Mobile Menu */}

   418→          <div className="lg:hidden py-4 border-t border-border/50">

   420→              {navItems.map(item => (

   422→                  key={item.id}
   423→                  onClick={() => {
   424→                    setActiveSection(item.id)
   425→                    setIsMobileOpen(false)

   428→                  className={`px-3 py-2 rounded-lg text-sm font-medium text-left transition-all ${

   431→                      : 'text-muted-foreground hover:text-foreground hover:bg-white/5'
   432→                  }`}
   433→                >
   434→                  {item.label}
   435→                </button>
   436→              ))}
   437→            </div>
   438→          </div>
   439→        )}
   440→      </div>
   441→    </nav>
   442→  )
   443→}

  1292→// Helper function for time formatting
  1293→function formatTimeAgo(date: Date): string {
  1294→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000)
  1295→  
  1296→  if (seconds < 60) return `${seconds}s ago`
  1297→  if (seconds < 3600) return `${Math.floor(seconds / 60)}m ago`
  1298→  if (seconds < 86400) return `${Math.floor(seconds / 3600)}h ago`
  1299→  return `${Math.floor(seconds / 86400)}d ago`
  1300→}
