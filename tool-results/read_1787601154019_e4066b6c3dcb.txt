   241→function ExpandableText({ content, maxLength = 150 }: { content: string; maxLength?: number }) {
   242→  const [expanded, setExpanded] = useState(false)
   243→  if (content.length <= maxLength) return <span className="text-sm">{content}</span>
   244→  return (
   245→    <span className="text-sm">
   246→      {!expanded ? `${content.substring(0, maxLength)}...` : content}
   247→      <button onClick={() => setExpanded(!expanded)} className="ml-1 text-primary hover:underline text-xs font-medium">
   248→        {expanded ? 'Show less' : 'Show more'}
   249→      </button>
   250→    </span>
   251→  )
   252→}

   279→function ChatWidget({ context, title }: {context: string; title: string}) {

   281→  const [messages, setMessages] = useState<ChatMessage[]>([

   297→      <Button onClick={() => setOpen(!open)} className="fixed bottom-6 right-6 z-50 rounded-full w-14 h-14 p-0 shadow-lg bg-primary text-primary-foreground">

   912→// Particle Background Component
   913→function ParticleCanvas() {
   914→  const canvasRef = useRef<HTMLCanvasElement>(null)
   915→  
   916→  useEffect(() => {
   917→    const canvas = canvasRef.current
   918→    if (!canvas) return
   919→    const ctx = canvas.getContext('2d')
   920→    if (!ctx) return
   921→    
   922→    let animId: number
   923→    const particles: Array<{x: number; y: number; size: number; vx: number; vy: number; opacity: number; pulse: number}> = []
   924→    
   925→    const resize = () => { canvas.width = window.innerWidth; canvas.height = window.innerHeight }
   926→    resize()
   927→    window.addEventListener('resize', resize)
   928→    
   929→    for (let i = 0; i < Math.min(Math.floor(window.innerWidth * 0.05), 60); i++) {
   930→      particles.push({
   931→        x: Math.random() * canvas.width, y: Math.random() * canvas.height,
   932→        size: Math.random() * 1.5 + 0.5, vx: (Math.random() - 0.5) * 0.3,
   933→        vy: (Math.random() - 0.5) * 0.3, opacity: Math.random() * 0.4 + 0.1,
   934→        pulse: Math.random() * Math.PI * 2
   935→      })
   936→    }
   937→    
   938→    const animate = () => {
   939→      ctx.clearRect(0, 0, canvas.width, canvas.height)
   940→      particles.forEach(p => {
   941→        p.x += p.vx; p.y += p.vy; p.pulse += 0.02
   942→        if (p.x < 0 || p.x > canvas.width || p.y < 0 || p.y > canvas.height) { p.x = Math.random() * canvas.width; p.y = Math.random() * canvas.height }
   943→        ctx.beginPath(); ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2)
   944→        ctx.fillStyle = `rgba(16, 185, 129, ${p.opacity + Math.sin(p.pulse) * 0.12})`; ctx.fill()
   945→      })
   946→      animId = requestAnimationFrame(animate)
   947→    }
   948→    animate()
   949→    return () => { window.removeEventListener('resize', resize); cancelAnimationFrame(animId) }
   950→  }, [])
   951→  
   952→  return <canvas ref={canvasRef} className="fixed inset-0 pointer-events-none z-0" style={{opacity: 0.5}} />
   953→}
