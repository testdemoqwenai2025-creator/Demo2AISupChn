   585→

   587→      result = result.filter(item => item.source === selectedSource)
   588→    }
   589→
   590→    if (selectedCategory !== 'all') {
   591→      result = result.filter(item => item.category === selectedCategory)
   592→    }

   594→    if (selectedImpact !== 'all') {

   599→      const query = searchQuery.toLowerCase()

   601→        item.title.toLowerCase().includes(query) ||
   602→        item.summary.toLowerCase().includes(query) ||
   603→        item.tags.some(tag => tag.toLowerCase().includes(query))
   604→      )
   605→    }
   606→

   613→      const next = new Set(prev)
   614→      if (next.has(id)) {
   615→        next.delete(id)
   616→      } else {
   617→        next.add(id)
   618→      }
   619→      return next
   620→    })

   628→    if (minutes < 60) return `${minutes}m ago`

   630→    if (hours < 24) return `${hours}h ago`
   631→    const days = Math.floor(hours / 24)
   632→    return `${days}d ago`
   633→  }
